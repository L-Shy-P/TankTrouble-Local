### 2.1 改动（`_tools/headless/cdp.js:145`）
改前：
```js
const userDataDir = opts.userDataDir || fs.mkdtempSync(path.join(os.tmpdir(), 'vt-chrome-'));
```
改后（**单行内联，避免 heredoc 事故**）：
```js
const VT_TMP = opts.tmpDir || process.env.VT_TMP_DIR || 'D:/vt_run';
let userDataDir = opts.userDataDir;
if (!userDataDir) {
  try { var _base = path.join(VT_TMP, 'chrome'); fs.mkdirSync(_base, { recursive: true });
        userDataDir = fs.mkdtempSync(path.join(_base, 'vt-chrome-')); }
  catch (eVT) { userDataDir = fs.mkdtempSync(path.join(os.tmpdir(), 'vt-chrome-')); }
}
```
要点：**默认 `D:/vt_run/chrome/vt-chrome-*`** · **两处可覆盖**（`opts.tmpDir` / 环境变量 `VT_TMP_DIR`）⟹ ⛔ 没把 D 写死成唯一选择 · **D 不可用自动回退** · `opts.userDataDir` 仍是最高优先（老调用方零影响）。
